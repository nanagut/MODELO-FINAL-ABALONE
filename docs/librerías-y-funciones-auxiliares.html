# Puntos que el diagrama de caja marca como atípicos (fuera de Q1 - 1.5*RIC y Q3 + 1.5*RIC)
contar_atipicos <- function(x) {
  q1 <- unname(quantile(x, 0.25)); q3 <- unname(quantile(x, 0.75)); r <- q3 - q1
  sum(x < q1 - 1.5 * r | x > q3 + 1.5 * r)
}

# Resumen descriptivo de una variable numérica
resumen <- function(x) {
  tibble(
    n = length(x), media = mean(x), sd = sd(x), mediana = median(x),
    RIC = IQR(x), Q1 = unname(quantile(x, 0.25)), Q3 = unname(quantile(x, 0.75)),
    minimo = min(x), maximo = max(x),
    asimetria = skewness(x), curtosis = kurtosis(x), atipicos = contar_atipicos(x)
  )
}
